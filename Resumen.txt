(function () {
  'use strict';
  var estado = { vista: null, respuesta: null, pais: '', segmento: '', orden: [], region: 0, leyendo: false };
  var MENSAJES = {
    FUENTE_SIN_HOJA: 'No se encontró alguna de las pestañas que usa la herramienta.',
    FUENTE_ESTRUCTURA: 'La estructura de la hoja cambió y no se pudo interpretar.',
    FUENTE_DEMASIADO_GRANDE: 'La tabla de datos supera el tamaño que la herramienta acepta leer.',
    FUENTE_ERROR_DE_SERVICIO: 'No se pudo leer el Sheets. Probá de nuevo en unos minutos.'
  };
  var DIAS = ['dom', 'lun', 'mar', 'mié', 'jue', 'vie', 'sáb'];
  var MESES = ['ene', 'feb', 'mar', 'abr', 'may', 'jun', 'jul', 'ago', 'sep', 'oct', 'nov', 'dic'];
  var NS = 'http://www.w3.org/2000/svg';
  function id(nombre) { return document.getElementById(nombre); }
  function crea(etiqueta, clase, texto) {
    var nodo = document.createElement(etiqueta);
    if (clase) nodo.className = clase;
    if (texto !== undefined && texto !== null) nodo.textContent = texto;
    return nodo;
  }
  function limpia(nodo) { while (nodo.firstChild) nodo.removeChild(nodo.firstChild); }
  function fechaCorta(iso) { var p = String(iso).split('-'); return p.length === 3 ? p[2] + '/' + p[1] : String(iso); }
  function fechaLarga(iso) { var p = String(iso).split('-'); return p.length === 3 ? p[2] + '/' + p[1] + '/' + p[0] : String(iso); }
  function fecha(iso) { return new Date(iso + 'T00:00:00Z'); }
  function diaMes(d) { return d.getUTCDate() + ' ' + MESES[d.getUTCMonth()]; }
  // "Semana 40 · 28 sep – 4 oct 2026" from the Monday that names the week.
  function rotuloSemana(iso) {
    var lunes = fecha(iso);
    if (isNaN(lunes.getTime())) return 'Semana del ' + iso;
    var domingo = new Date(lunes.getTime() + 6 * 86400000);
    var jueves = new Date(lunes.getTime() + 3 * 86400000);
    var inicio = Date.UTC(jueves.getUTCFullYear(), 0, 1);
    var numero = Math.floor((jueves.getTime() - inicio) / (7 * 86400000)) + 1;
    return 'Semana ' + numero + ' · ' + diaMes(lunes) + ' – ' + diaMes(domingo) + ' ' + domingo.getUTCFullYear();
  }
  function tono(favorable) { return favorable === true ? 'bueno' : favorable === false ? 'malo' : 'neutro'; }
  function flecha(sentido) { return sentido === 'sube' ? '▲ ' : sentido === 'baja' ? '▼ ' : ''; }
  function pastillaCambio(item) {
    return crea('span', 'pastilla ' + tono(item.favorable), item.cambio ? flecha(item.sentido) + item.cambio : 'Dato a revisar');
  }

  function leer(forzar) {
    if (estado.leyendo) return;
    estado.leyendo = true;
    id('actualizar').disabled = true;
    id('estado-lectura').textContent = forzar
      ? 'Leyendo y analizando el Sheets de nuevo. Tarda unos 15 segundos…'
      : 'Buscando la última corrida. Si hay que leer el Sheets, tarda unos 15 segundos…';
    google.script.run.withSuccessHandler(recibe).withFailureHandler(function () {
      recibe({ estado: 'error', error: { codigo: 'FUENTE_ERROR_DE_SERVICIO' } });
    }).obtenerVistaWeekly(forzar);
  }

  function recibe(respuesta) {
    estado.leyendo = false;
    id('actualizar').disabled = false;
    id('estado-lectura').textContent = '';
    var valida = respuesta && respuesta.schemaVersion === 'weekly-vista/3' && respuesta.estado === 'ok' && respuesta.vista;
    if (!valida) {
      var error = (respuesta && respuesta.error) || {};
      var caja = id('error');
      caja.textContent = (MENSAJES[error.codigo] || MENSAJES.FUENTE_ERROR_DE_SERVICIO) +
        (error.detalle && error.codigo !== 'FUENTE_ERROR_DE_SERVICIO' ? ' Detalle: ' + error.detalle : '');
      caja.hidden = false;
      if (estado.vista) {
        id('anterior').textContent = 'Lo que se ve abajo es el resultado anterior, leído el ' +
          new Date(estado.respuesta.generadoIso).toLocaleString('es-AR') + '. No está actualizado.';
        id('anterior').hidden = false;
      } else {
        id('subtitulo').className = 'linea';
        id('subtitulo').textContent = 'Sin resultado para mostrar.';
      }
      return;
    }
    id('error').hidden = true;
    id('anterior').textContent = (respuesta.avisos || []).join(' ');
    id('anterior').hidden = !(respuesta.avisos || []).length;
    estado.vista = respuesta.vista;
    estado.respuesta = respuesta;
    estado.orden = [];
    pintaTodo();
  }

  function pintaTodo() {
    var vista = estado.vista;
    var respuesta = estado.respuesta;
    id('contenido').hidden = false;
    id('subtitulo').className = 'linea';
    id('subtitulo').textContent = 'Semana del ' + fechaLarga(vista.semana) + ' comparada con la del ' +
      fechaLarga(vista.semanaAnterior) + ' · ' + vista.semanasDisponibles + ' semanas de historia';
    id('sello-semana').textContent = rotuloSemana(vista.semana);
    id('sello-semana').hidden = false;
    id('sello-datos').textContent = 'Datos del Sheets · actualizados el ' +
      (respuesta.origen.ultimaActualizacionExtract ? fechaLarga(respuesta.origen.ultimaActualizacionExtract) +
        (respuesta.origen.ultimaActualizacionHora ? ', ' + respuesta.origen.ultimaActualizacionHora : '') : 'fecha no informada');
    id('sello-datos').className = 'pastilla ' + (vista.provisoria ? 'aviso' : 'bueno');

    var carga = id('carga');

      carga.appendChild(crea('div', 'aviso-caja error', item.titulo + '. ' + item.detalle));
    });
    lista(id('limites'), vista.limites);
    lista(id('advertencias'), vista.advertencias);

      (vista.conHistorial ? '.' : '. Sin corrida guardada de la semana anterior: todavía no se puede marcar qué es nuevo.');

    pintaRevisar();

  // The headline KPIs of LATAM as buttons over one chart: the chosen one
  // (estado.region, Orders first) is the one charted.
  function pintaRegion(region, semanas, foco) {
    var topline = id('region-topline');
    limpia(topline);
    if (!(estado.region >= 0 && estado.region < region.topline.length)) estado.region = 0;
    region.topline.forEach(function (item, i) {
      var boton = crea('button', 'tarjeta');
      boton.type = 'button';
      boton.setAttribute('aria-pressed', String(i === estado.region));
      boton.title = 'Ver el gráfico de ' + item.nombre;
      boton.appendChild(crea('div', 'nombre', item.nombre));
      var fila = crea('div', 'fila-valor');
      fila.appendChild(crea('span', 'grande', item.valor));
      boton.appendChild(fila);
      boton.appendChild(pastillaCambio(item));
      boton.addEventListener('click', function () { estado.region = i; pintaRegion(region, semanas, i); });
      topline.appendChild(boton);
    });
    if (foco !== undefined && topline.children[foco]) topline.children[foco].focus();
    pintaCaja('region-grafico', region.topline[estado.region] || null, region.nombre, semanas);
  }

  // Fills one chart box (the region's or the market's): title, change, the
  // line under it, the legend and the chart. Returns the title.
  function pintaCaja(prefijo, elegido, entidad, semanas) {
    var g = elegido && elegido.grafico;
    var caja = id(prefijo);
    limpia(caja);
    var titulo = elegido ? (elegido.contexto && elegido.contexto.toLowerCase() !== elegido.nombre.toLowerCase()
      ? elegido.contexto + ' › ' : '') + elegido.nombre : 'Sin datos para graficar';
    id(prefijo + '-titulo').textContent = titulo;
    var cambio = id(prefijo + '-cambio');
    limpia(cambio);
    if (elegido) cambio.appendChild(pastillaCambio(elegido));
    id(prefijo + '-bajada').textContent = g ? entidad + ' · últimas ' + g.valores.length + ' semanas (lunes de cada semana) · fila ' + elegido.fila : '';
    id(prefijo + '-leyenda').textContent = g && g.promedioTexto
      ? '- - - Promedio previo (' + g.semanasPromedio + ' sem.): ' + g.promedioTexto : '';
    if (g) caja.appendChild(grafico(elegido, semanas));
    return titulo;
  }

  function nombreKpi(item) {
    var nombre = crea('div', 'nombre', item.nombre);
    if (item.contexto && item.contexto.toLowerCase() !== item.nombre.toLowerCase()) nombre.appendChild(crea('small', '', item.contexto));
    return nombre;
  }

  function tarjetaMovida(item) {
    var tarjeta = crea('div', 'tarjeta movida');
    tarjeta.appendChild(nombreKpi(item));
    var fila = crea('div', 'fila-valor');
    fila.appendChild(crea('span', 'grande', item.valor));
    fila.appendChild(pastillaCambio(item));
    tarjeta.appendChild(fila);
    tarjeta.appendChild(crea('hr'));
    tarjeta.appendChild(crea('span', 'rotulo dato', 'Lo que vemos · dato'));
    tarjeta.appendChild(crea('p', '', item.dato));
    if (item.hipotesis) {
      tarjeta.appendChild(crea('span', 'rotulo causa', 'Posible causa · hipótesis a validar'));
      tarjeta.appendChild(crea('p', 'gris', item.hipotesis));
    }
    return tarjeta;
  }

  function pintaHistoria() {
    var vista = estado.vista;
    var historia = vista.historia;
    var hay = !!historia;
    id('capitulo-region').hidden = !hay || !historia.regional;
    id('capitulo-pais').hidden = !hay || !historia.paises.length;
    if (!hay) return;
    var region = historia.regional;
    if (region) {
      id('region-titular').textContent = region.titular;
      id('region-texto').textContent = region.texto;
      pintaRegion(region, historia.semanasGrafico);
      var movidas = id('movidas');
      limpia(movidas);
      region.movidas.forEach(function (item) { movidas.appendChild(tarjetaMovida(item)); });
      if (!region.movidas.length) movidas.appendChild(crea('div', 'vacio', 'Ningún KPI de LATAM se movió fuera de lo habitual esta semana.'));
    }
    var fichas = id('paises');
    limpia(fichas);
    if (!historia.paises.some(function (p) { return p.nombre === estado.pais; })) estado.pais = historia.paisInicial || '';
    historia.paises.forEach(function (pais) {
      var boton = crea('button', '', pais.nombre);
      boton.type = 'button';
      boton.setAttribute('aria-pressed', String(pais.nombre === estado.pais));
      boton.addEventListener('click', function () { estado.pais = pais.nombre; estado.orden = []; pintaHistoria(); });
      fichas.appendChild(boton);
    });
    var pais = historia.paises.filter(function (p) { return p.nombre === estado.pais; })[0];
    if (pais) pintaPais(pais, vista.semana, historia.semanasGrafico);

    id('cierre-titular').textContent = historia.cierre.titular;
    var pasos = id('cierre-pasos');
    limpia(pasos);
    historia.cierre.pasos.forEach(function (paso, i) {
      var columna = crea('div');
      var titulo = crea('h3');
      titulo.appendChild(crea('span', '', String(i + 1)));
      titulo.appendChild(document.createTextNode(paso.titulo));
      columna.appendChild(titulo);
      if (paso.lista) {
        var ol = crea('ol');
        paso.lista.forEach(function (texto) { ol.appendChild(crea('li', '', texto)); });
        columna.appendChild(ol);
      } else {
        columna.appendChild(crea('p', '', paso.texto));
      }
      pasos.appendChild(columna);
    });
    var respuesta = estado.respuesta;
    var chips = id('cierre-estado');
    limpia(chips);
    chips.appendChild(crea('span', 'rotulo', 'Estado de la información'));
    ['Última actualización: ' + (respuesta.origen.ultimaActualizacionExtract
        ? diaSemana(respuesta.origen.ultimaActualizacionExtract) +
          (respuesta.origen.ultimaActualizacionHora ? ', ' + respuesta.origen.ultimaActualizacionHora : '') : 'no informada'),
      'Cobertura: ' + historia.paises.length + ' países' + (region ? ' + LATAM' : ''),
      historia.totalCalidad === 1 ? '1 alerta de calidad de datos abierta' : historia.totalCalidad + ' alertas de calidad de datos abiertas',
      'Las causas son hipótesis, no confirmaciones'].forEach(function (texto) {
      chips.appendChild(crea('span', 'pastilla', texto));
    });
  }

  // "lun 5 oct" from an ISO date.
  function diaSemana(iso) {
    var d = fecha(iso);
    return isNaN(d.getTime()) ? String(iso) : DIAS[d.getUTCDay()] + ' ' + diaMes(d);
  }

  // One market: its text, the segments of the sheet as buttons, the chart of
  // one KPI of the chosen segment and the others as cards. The KPI that moved
  // the most opens in the chart; choosing a card swaps places with it, and the
  // text on the left always tells about the KPI in the chart. estado.orden
  // holds which KPI sits in the chart (first) and in each card. The chosen
  // segment is kept from one market to the next.
  function pintaPais(pais, semana, semanas, foco) {
    id('pais-nombre').textContent = pais.nombre;
    id('pais-semana').textContent = rotuloSemana(semana);
    id('pais-texto').textContent = pais.texto;
    id('pais-nota').textContent = pais.nota || '';
    id('pais-nota').hidden = !pais.nota;
    var segmento = pais.segmentos.filter(function (s) { return s.nombre === estado.segmento; })[0] || pais.segmentos[0];
    estado.segmento = segmento.nombre;
    var botones = id('segmentos');
    limpia(botones);
    pais.segmentos.forEach(function (s) {
      var boton = crea('button', '', s.nombre);
      boton.type = 'button';
      boton.setAttribute('aria-pressed', String(s === segmento));
      if (s.movidos) {
        boton.appendChild(crea('small', '', String(s.movidos)));
        boton.title = s.movidos + ' de ' + s.total + ' KPIs se movieron por encima de su variación habitual';
      }
      boton.addEventListener('click', function () {
        estado.segmento = s.nombre;
        estado.orden = [];
        pintaPais(pais, semana, semanas);
      });
      botones.appendChild(boton);
    });
    id('segmento-texto').textContent = segmento.texto;
    var kpis = segmento.kpis;
    if (estado.orden.length !== kpis.length) estado.orden = kpis.map(function (kpi, i) { return i; });
    var elegido = kpis[estado.orden[0]] || null;
    var titulo = pintaCaja('grafico', elegido, pais.nombre, semanas);
    id('kpi-elegido').textContent = elegido ? titulo : '';
    id('kpi-dato').textContent = elegido ? elegido.dato : 'Este segmento no tiene KPIs con datos en ' + pais.nombre + '.';
    id('kpi-rotulo-causa').hidden = !elegido || !elegido.hipotesis;
    id('kpi-hipotesis').textContent = elegido && elegido.hipotesis ? elegido.hipotesis : '';
    var tarjetas = id('pais-tarjetas');
    limpia(tarjetas);
    id('pais-rotulo').hidden = kpis.length < 2;
    id('pais-rotulo').textContent = segmento.general ? 'Otros KPIs del país esta semana' : 'Otros KPIs de ' + segmento.nombre + ' esta semana';
    estado.orden.slice(1).forEach(function (indice, i) {
      var item = kpis[indice];
      var boton = crea('button', 'tarjeta');
      boton.type = 'button';
      boton.appendChild(nombreKpi(item));
      boton.appendChild(crea('span', 'grande', item.valor));
      boton.appendChild(pastillaCambio(item));
      boton.appendChild(crea('span', 'previo', 'Sem. anterior: ' + item.anterior));
      boton.appendChild(crea('span', 'ver', 'Ver su gráfico'));
      boton.addEventListener('click', function () {
        estado.orden[i + 1] = estado.orden[0];
        estado.orden[0] = indice;
        pintaPais(pais, semana, semanas, i);
      });
      tarjetas.appendChild(boton);
    });
    if (foco !== undefined && tarjetas.children[foco]) tarjetas.children[foco].focus();
  }

  function svg(nombre, atributos, texto) {
    var nodo = document.createElementNS(NS, nombre);
    Object.keys(atributos).forEach(function (clave) { nodo.setAttribute(clave, atributos[clave]); });
    if (texto !== undefined) nodo.textContent = texto;
    return nodo;
  }

  // Line of the compared weeks with the average of the previous ones. A week
  // without data stays a gap in the line. The weeks are the same for every
  // KPI and arrive once; each KPI brings its values and their texts.
  function grafico(item, semanas) {
    var g = { nombre: item.nombre, favorable: item.favorable, cambio: item.cambio, promedioPrevio: item.grafico.promedioPrevio,
      serie: item.grafico.valores.map(function (valor, i) {
        return { semana: semanas[i], valor: valor, texto: item.grafico.textos[i] };
      }) };
    var ancho = 760, alto = 250, izq = 64, der = 24, arriba = 18, abajo = 34;
    var lienzo = svg('svg', { 'class': 'grafico', viewBox: '0 0 ' + ancho + ' ' + alto, role: 'img',
      'aria-label': g.nombre + ': ' + g.serie.map(function (p) { return fechaCorta(p.semana) + ' ' + p.texto; }).join(', ') });
    var conocidos = g.serie.filter(function (p) { return p.valor !== null; });
    if (conocidos.length < 2) {
      lienzo.appendChild(svg('text', { x: ancho / 2, y: alto / 2, 'text-anchor': 'middle', fill: '#6b6461', 'font-size': 13 },
        'No hay suficientes semanas con dato para graficar.'));
      return lienzo;
    }
    var valores = conocidos.map(function (p) { return p.valor; });
    if (g.promedioPrevio !== null) valores.push(g.promedioPrevio);
    var minimo = Math.min.apply(null, valores), maximo = Math.max.apply(null, valores);
    var holgura = (maximo - minimo || Math.abs(maximo) || 1) * 0.18;
    minimo -= holgura; maximo += holgura;
    var x = function (i) { return izq + i * (ancho - izq - der) / (g.serie.length - 1); };
    var y = function (v) { return arriba + (maximo - v) * (alto - arriba - abajo) / (maximo - minimo); };
    var base = alto - abajo;
    var menor = conocidos.reduce(function (a, b) { return b.valor < a.valor ? b : a; });
    var mayor = conocidos.reduce(function (a, b) { return b.valor > a.valor ? b : a; });
    [mayor, menor].forEach(function (p) {
      lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: y(p.valor), y2: y(p.valor), stroke: '#efe6e1', 'stroke-width': 1 }));
      lienzo.appendChild(svg('text', { x: izq - 10, y: y(p.valor) + 4, 'text-anchor': 'end', fill: '#6b6461', 'font-size': 11 }, p.texto));
    });
    lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: base, y2: base, stroke: '#e3d9d3', 'stroke-width': 1 }));
    var tramos = [];
    var tramo = [];
    g.serie.forEach(function (p, i) {
      if (p.valor === null) { if (tramo.length) tramos.push(tramo); tramo = []; return; }
      tramo.push([x(i), y(p.valor)]);
    });
    if (tramo.length) tramos.push(tramo);
    tramos.forEach(function (puntos) {
      if (puntos.length < 2) return;
      var linea = puntos.map(function (p) { return p[0].toFixed(1) + ',' + p[1].toFixed(1); }).join(' ');
      lienzo.appendChild(svg('polygon', { points: puntos[0][0].toFixed(1) + ',' + base + ' ' + linea + ' ' +
        puntos[puntos.length - 1][0].toFixed(1) + ',' + base, fill: '#fa0050', 'fill-opacity': '.12' }));
      lienzo.appendChild(svg('polyline', { points: linea, fill: 'none', stroke: '#fa0050', 'stroke-width': 2.5,
        'stroke-linejoin': 'round', 'stroke-linecap': 'round' }));
    });
    if (g.promedioPrevio !== null) {
      lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: y(g.promedioPrevio), y2: y(g.promedioPrevio),
        stroke: '#8f8782', 'stroke-width': 1.5, 'stroke-dasharray': '5 5' }));
    }
    g.serie.forEach(function (p, i) {
      var ultima = i === g.serie.length - 1;
      lienzo.appendChild(svg('text', { x: x(i), y: alto - 10, 'text-anchor': 'middle', fill: ultima ? '#1f1d1d' : '#6b6461',
        'font-size': 11, 'font-weight': ultima ? 800 : 400 }, diaMes(fecha(p.semana))));
      if (p.valor === null) return;
      lienzo.appendChild(svg('circle', { cx: x(i), cy: y(p.valor), r: ultima ? 5 : 3.5, fill: ultima ? '#fa0050' : '#ffffff',
        stroke: '#fa0050', 'stroke-width': 2 }));
    });
    var final = g.serie[g.serie.length - 1];
    if (final.valor !== null) {
      var arribaDelPunto = y(final.valor) > arriba + 26;
      lienzo.appendChild(svg('text', { x: ancho - der - 12, y: y(final.valor) + (arribaDelPunto ? -12 : 20), 'text-anchor': 'end',
        fill: g.favorable === true ? '#14934f' : g.favorable === false ? '#d21f3c' : '#1f1d1d', 'font-size': 12, 'font-weight': 800,
        stroke: '#faf3ef', 'stroke-width': 4, 'paint-order': 'stroke' }, final.texto + (g.cambio ? ' · ' + g.cambio + ' vs sem. anterior' : '')));
    }
    return lienzo;
  }

  // ---- Data to review ----------------------------------------------------------------

  function lista(nodo, textos) { limpia(nodo); textos.forEach(function (t) { nodo.appendChild(crea('li', '', t)); }); }

  // One line per rule and KPI. marcas turns the count of marked markets into
  // words: [all, none, some].
  function renglon(grupo, marcas) {
    var n = grupo.entidades.length;
    var fila = crea('div', 'renglon');
    fila.appendChild(crea('span', 'donde', n === 1 ? grupo.entidades[0] : n + ' mercados'));
    var medio = crea('div');
    medio.appendChild(crea('strong', '', grupo.kpi));
    medio.appendChild(crea('small', '', grupo.titulo + '. Fila ' + grupo.fila + '.' +
      (n > 1 ? ' ' + grupo.entidades.join(', ') + '.' : grupo.detalle ? ' ' + grupo.detalle : '')));
    fila.appendChild(medio);
    var lado = crea('div', 'lado');
    lado.appendChild(crea('span', 'pastilla ' + (grupo.severidad === 'alta' ? 'malo' : grupo.severidad === 'media' ? 'aviso' : 'neutro'),
      'Prioridad ' + grupo.severidad));
    if (grupo.marcados !== null) {
      lado.appendChild(crea('small', '', grupo.marcados === n ? marcas[0] : grupo.marcados === 0 ? marcas[1]
        : marcas[2].replace('{n}', grupo.marcados).replace('{de}', n)));
    }
    fila.appendChild(lado);
    return fila;
  }

  function pintaRevisar() {
    var vista = estado.vista;
    var revisar = vista.revisar || { actual: [], anterior: null };
    var actual = id('revisar-actual');
    limpia(actual);
    id('titulo-revisar-actual').textContent = revisar.actual.length + ' · ' + rotuloSemana(vista.semana);
    revisar.actual.forEach(function (grupo) {
      actual.appendChild(renglon(grupo, ['Nuevo esta semana', 'También la semana pasada', 'Nuevo en {n} de {de}']));
    });
    if (!revisar.actual.length) actual.appendChild(crea('div', 'vacio', 'Esta semana no se detectaron datos a revisar.'));
    var anterior = id('revisar-anterior');
    limpia(anterior);
    id('titulo-revisar-anterior').textContent = (revisar.anterior ? revisar.anterior.grupos.length + ' · ' : '') + rotuloSemana(vista.semanaAnterior);
    if (!revisar.anterior) {
      anterior.appendChild(crea('div', 'vacio', 'No hay una corrida guardada de esa semana en la hoja de registro. ' +
        'La lista aparece cuando la herramienta ya analizó y guardó la semana anterior.'));
      return;
    }
    revisar.anterior.grupos.forEach(function (grupo) {
      anterior.appendChild(renglon(grupo, ['Sigue esta semana', 'Ya no aparece', 'Sigue en {n} de {de}']));
    });
    if (!revisar.anterior.grupos.length) anterior.appendChild(crea('div', 'vacio', 'La semana pasada no hubo datos a revisar.'));
  }

  id('actualizar').addEventListener('click', function () { leer(true); });
  leer(false);
