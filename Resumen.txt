    var inicio = Date.UTC(jueves.getUTCFullYear(), 0, 1);
    var numero = Math.floor((jueves.getTime() - inicio) / (7 * 86400000)) + 1;
    return 'Semana ' + numero + ' · ' + diaMes(lunes) + ' – ' + diaMes(domingo) + ' ' + domingo.getUTCFullYear();
  }
  function tono(favorable) { return favorable === true ? 'bueno' : favorable === false ? 'malo' : 'neutro'; }

  function leer(forzar) {
    if (estado.leyendo) return;
    estado.leyendo = true;
    id('actualizar').disabled = true;
    id('estado-lectura').textContent = forzar
      ? 'Leyendo y analizando el Sheets de nuevo. Tarda unos 15 segundos…'
      : 'Buscando la última corrida. Si hay que leer el Sheets, tarda unos 15 segundos…';
    google.script.run.withSuccessHandler(recibe).withFailureHandler(function () {
      recibe({ estado: 'error', error: { codigo: 'FUENTE_ERROR_DE_SERVICIO' } });
    }).obtenerVistaWeekly(forzar);

  function recibe(respuesta) {
    estado.leyendo = false;
    id('actualizar').disabled = false;
    id('estado-lectura').textContent = '';
    var valida = respuesta && respuesta.schemaVersion === 'weekly-vista/2' && respuesta.estado === 'ok' && respuesta.vista;
    if (!valida) {
      var error = (respuesta && respuesta.error) || {};
      var caja = id('error');
      caja.textContent = (MENSAJES[error.codigo] || MENSAJES.FUENTE_ERROR_DE_SERVICIO) +
        (error.detalle && error.codigo !== 'FUENTE_ERROR_DE_SERVICIO' ? ' Detalle: ' + error.detalle : '');
      caja.hidden = false;
      if (estado.vista) {
        id('anterior').textContent = 'Lo que se ve abajo es el resultado anterior, leído el ' +
          new Date(estado.respuesta.generadoIso).toLocaleString('es-AR') + '. No está actualizado.';
        id('anterior').hidden = false;
      } else {
        id('subtitulo').className = 'linea';
        id('subtitulo').textContent = 'Sin resultado para mostrar.';
      }
      return;
    }
    id('error').hidden = true;
    id('anterior').textContent = (respuesta.avisos || []).join(' ');
    id('anterior').hidden = !(respuesta.avisos || []).length;
    estado.vista = respuesta.vista;
    estado.respuesta = respuesta;
    pintaTodo();
  }

  function pintaLamina() {
    var todas = laminas();
    if (!todas.length) return;
    estado.lamina = Math.max(0, Math.min(estado.lamina, todas.length - 1));
    var pista = id('lamina-pista');
    pista.style.transform = 'translateX(' + (-100 * estado.lamina) + '%)';
    Array.prototype.forEach.call(pista.children, function (hoja, i) {
      // Slides out of view must not be reachable by keyboard or screen reader.
      if (i === estado.lamina) { hoja.removeAttribute('inert'); hoja.removeAttribute('aria-hidden'); }
      else { hoja.setAttribute('inert', ''); hoja.setAttribute('aria-hidden', 'true'); }
    });
    // The strip is as tall as its tallest slide; the window follows the slide in view.
    id('lamina-visor').style.height = pista.children[estado.lamina].offsetHeight + 'px';
    id('lamina-titulo').textContent = todas[estado.lamina].seccion;
    id('lamina-cuenta').textContent = 'Sección ' + (estado.lamina + 1) + ' de ' + todas.length + ' · LATAM';
    id('lamina-anterior').disabled = estado.lamina === 0;
    id('lamina-siguiente').disabled = estado.lamina === todas.length - 1;
    Array.prototype.forEach.call(id('lamina-puntos').children, function (boton, i) {
      boton.setAttribute('aria-pressed', String(i === estado.lamina));
    });
  }
  function mueveLamina(paso) { estado.lamina += paso; pintaLamina(); }

  // Dragging or swiping the strip moves one slide; a short drag springs back.
  // One gesture is one pointer from press to release, and moves one slide at most.
  function arrastreLaminas() {
    var visor = id('lamina-visor');
    var pista = id('lamina-pista');
    var gesto = null;
    visor.addEventListener('pointerdown', function (e) {
      if (gesto || (e.pointerType === 'mouse' && e.button !== 0)) return;
      gesto = { puntero: e.pointerId, desde: e.clientX, arrastra: false };
    });
    visor.addEventListener('pointermove', function (e) {
      if (!gesto || e.pointerId !== gesto.puntero) return;
      var corrido = e.clientX - gesto.desde;
      if (!gesto.arrastra && Math.abs(corrido) < 8) return;
      if (!gesto.arrastra) {
        gesto.arrastra = true;
        visor.classList.add('arrastra');
        // Keeps the gesture alive when the pointer leaves the strip; not every pointer can be captured.
        try { visor.setPointerCapture(e.pointerId); } catch (error) { /* the gesture still ends on release */ }
        var seleccion = window.getSelection && window.getSelection();
        if (seleccion) seleccion.removeAllRanges();
      }
      e.preventDefault();
      pista.style.transform = 'translateX(calc(' + (-100 * estado.lamina) + '% + ' + corrido + 'px))';
    });
    var suelta = function (e) {
      if (!gesto || e.pointerId !== gesto.puntero) return;
      var corrido = gesto.arrastra ? e.clientX - gesto.desde : 0;
      gesto = null;
      visor.classList.remove('arrastra');
      var umbral = Math.min(120, visor.clientWidth / 6);
      if (corrido <= -umbral) estado.lamina += 1;
      if (corrido >= umbral) estado.lamina -= 1;
      pintaLamina();
    };
    visor.addEventListener('pointerup', suelta);
    visor.addEventListener('pointercancel', suelta);
  }

  function svg(nombre, atributos, texto) {
    var nodo = document.createElementNS(NS, nombre);
    Object.keys(atributos).forEach(function (clave) { nodo.setAttribute(clave, atributos[clave]); });
    if (texto !== undefined) nodo.textContent = texto;
    return nodo;
  }

  // Line of the compared weeks with the average of the previous ones. A week
  // without data stays a gap in the line.
  function grafico(g) {
    var ancho = 760, alto = 250, izq = 64, der = 24, arriba = 18, abajo = 34;
    var lienzo = svg('svg', { 'class': 'grafico', viewBox: '0 0 ' + ancho + ' ' + alto, role: 'img',
      'aria-label': g.nombre + ': ' + g.serie.map(function (p) { return fechaCorta(p.semana) + ' ' + p.texto; }).join(', ') });
    var conocidos = g.serie.filter(function (p) { return p.valor !== null; });
    if (conocidos.length < 2) {
      lienzo.appendChild(svg('text', { x: ancho / 2, y: alto / 2, 'text-anchor': 'middle', fill: '#6b6461', 'font-size': 13 },
        'No hay suficientes semanas con dato para graficar.'));
      return lienzo;
    }
    var valores = conocidos.map(function (p) { return p.valor; });
    if (g.promedioPrevio !== null) valores.push(g.promedioPrevio);
    var minimo = Math.min.apply(null, valores), maximo = Math.max.apply(null, valores);
    var holgura = (maximo - minimo || Math.abs(maximo) || 1) * 0.18;
    minimo -= holgura; maximo += holgura;
    var x = function (i) { return izq + i * (ancho - izq - der) / (g.serie.length - 1); };
    var y = function (v) { return arriba + (maximo - v) * (alto - arriba - abajo) / (maximo - minimo); };
    var base = alto - abajo;
    var menor = conocidos.reduce(function (a, b) { return b.valor < a.valor ? b : a; });
    var mayor = conocidos.reduce(function (a, b) { return b.valor > a.valor ? b : a; });
    [mayor, menor].forEach(function (p) {
      lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: y(p.valor), y2: y(p.valor), stroke: '#efe6e1', 'stroke-width': 1 }));
      lienzo.appendChild(svg('text', { x: izq - 10, y: y(p.valor) + 4, 'text-anchor': 'end', fill: '#6b6461', 'font-size': 11 }, p.texto));
    });
    lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: base, y2: base, stroke: '#e3d9d3', 'stroke-width': 1 }));
    var tramos = [];
    var tramo = [];
    g.serie.forEach(function (p, i) {
      if (p.valor === null) { if (tramo.length) tramos.push(tramo); tramo = []; return; }
      tramo.push([x(i), y(p.valor)]);
    });
    if (tramo.length) tramos.push(tramo);
    tramos.forEach(function (puntos) {
      if (puntos.length < 2) return;
      var linea = puntos.map(function (p) { return p[0].toFixed(1) + ',' + p[1].toFixed(1); }).join(' ');
      lienzo.appendChild(svg('polygon', { points: puntos[0][0].toFixed(1) + ',' + base + ' ' + linea + ' ' +
        puntos[puntos.length - 1][0].toFixed(1) + ',' + base, fill: '#fa0050', 'fill-opacity': '.12' }));
      lienzo.appendChild(svg('polyline', { points: linea, fill: 'none', stroke: '#fa0050', 'stroke-width': 2.5,
        'stroke-linejoin': 'round', 'stroke-linecap': 'round' }));
    });
    if (g.promedioPrevio !== null) {
      lienzo.appendChild(svg('line', { x1: izq, x2: ancho - der, y1: y(g.promedioPrevio), y2: y(g.promedioPrevio),
        stroke: '#8f8782', 'stroke-width': 1.5, 'stroke-dasharray': '5 5' }));
    }
    g.serie.forEach(function (p, i) {
      var ultima = i === g.serie.length - 1;
      lienzo.appendChild(svg('text', { x: x(i), y: alto - 10, 'text-anchor': 'middle', fill: ultima ? '#1f1d1d' : '#6b6461',
        'font-size': 11, 'font-weight': ultima ? 800 : 400 }, diaMes(fecha(p.semana))));
      if (p.valor === null) return;
      lienzo.appendChild(svg('circle', { cx: x(i), cy: y(p.valor), r: ultima ? 5 : 3.5, fill: ultima ? '#fa0050' : '#ffffff',
        stroke: '#fa0050', 'stroke-width': 2 }));
    });
    var final = g.serie[g.serie.length - 1];
    if (final.valor !== null) {
      var arribaDelPunto = y(final.valor) > arriba + 26;
      lienzo.appendChild(svg('text', { x: ancho - der - 12, y: y(final.valor) + (arribaDelPunto ? -12 : 20), 'text-anchor': 'end',
        fill: '#d21f3c', 'font-size': 12, 'font-weight': 800,
        stroke: '#faf3ef', 'stroke-width': 4, 'paint-order': 'stroke' }, final.texto + (g.cambio ? ' · ' + g.cambio + ' vs sem. anterior' : '')));
    }
    return lienzo;
  }

  function opcion(valor, texto) { var o = crea('option', '', texto); o.value = valor; return o; }
  function lista(nodo, textos) { limpia(nodo); textos.forEach(function (t) { nodo.appendChild(crea('li', '', t)); }); }

  function pasa(item) {
    if (estado.mercado && item.entidad !== estado.mercado) return false;
    if (estado.texto) {
      var donde = (item.kpi + ' ' + (item.clave || '') + ' ' + (item.titulo || '')).toLowerCase();
      if (donde.indexOf(estado.texto) === -1) return false;
    }
    if (estado.pestana === 'movimientos' && estado.sentido) {
      if (estado.sentido === 'desfavorable' && item.favorable !== false) return false;
      if (estado.sentido === 'favorable' && item.favorable !== true) return false;
    }
    return true;
  }

  function pintaLista() {
    var vista = estado.vista;
    var movimientos = estado.pestana === 'movimientos';
    id('tab-movimientos').setAttribute('aria-selected', String(movimientos));
    id('tab-calidad').setAttribute('aria-selected', String(!movimientos));
    id('filtro-sentido').disabled = !movimientos;
    var items = (movimientos ? vista.movimientos : vista.calidad).filter(pasa);
    var contenedor = id('lista');
    limpia(contenedor);
    if (!items.length) {
      contenedor.appendChild(crea('div', 'vacio', movimientos
        ? 'Sin movimientos fuera de lo habitual para los filtros elegidos.'
        : 'Sin datos a revisar para los filtros elegidos.'));
      return;
    }
    items.forEach(function (item) { contenedor.appendChild(movimientos ? filaMovimiento(item) : filaCalidad(item)); });
  }

  function cabecera(item, derecha) {
    var resumen = crea('summary');
    var mercado = crea('div', 'mercado', item.entidad);
    if (item.esRegional) mercado.appendChild(crea('small', '', 'Agregado regional'));
    resumen.appendChild(mercado);
    var kpi = crea('div', 'kpi', item.kpi);
    kpi.appendChild(crea('small', '', 'Fila ' + item.fila + (item.clave ? ' · ' + item.clave : '') +
      (item.historial === 'nueva' ? ' · Nuevo esta semana' : item.historial === 'continua' ? ' · También la semana anterior' : '')));
    resumen.appendChild(kpi);
    resumen.appendChild(derecha);
    resumen.appendChild(chispa(item.serie));
    return resumen;
  }

  function filaMovimiento(item) {
    var caja = crea('details', 'item');
    var color = item.favorable === true ? 'bueno' : item.favorable === false ? 'malo' : '';
    var cambio = crea('div', 'cambio ' + color, flecha(item.sentido) + item.cambio);
    cambio.appendChild(crea('small', '', item.de + ' → ' + item.a));
    caja.appendChild(cabecera(item, cambio));
    var cuerpo = crea('div', 'cuerpo');
    cuerpo.appendChild(crea('span', 'pastilla ' + tono(item.favorable),
      item.favorable === true ? 'Favorable' : item.favorable === false ? 'Desfavorable' : 'Sin dirección definida'));
    cuerpo.appendChild(crea('span', 'pastilla neutro', item.medida));
    if (item.explicaciones.length) {
      cuerpo.appendChild(crea('h3', '', 'Qué más muestran los datos (hipótesis)'));
      var ul = crea('ul');
      item.explicaciones.forEach(function (texto) { ul.appendChild(crea('li', '', texto)); });
      cuerpo.appendChild(ul);
    }
    cuerpo.appendChild(tabla(item.serie));
    caja.appendChild(cuerpo);
    return caja;
  }

  function filaCalidad(item) {
    var caja = crea('details', 'item');
    var clase = item.severidad === 'alta' ? 'malo' : item.severidad === 'media' ? 'aviso' : 'neutro';
    var derecha = crea('div', 'cambio');
    derecha.appendChild(crea('span', 'pastilla ' + clase, item.titulo));
    caja.appendChild(cabecera(item, derecha));
    var cuerpo = crea('div', 'cuerpo');
    cuerpo.appendChild(crea('span', 'pastilla ' + clase, 'Prioridad ' + item.severidad));
    cuerpo.appendChild(crea('p', '', item.detalle));
    if (item.serie.length) cuerpo.appendChild(tabla(item.serie));
    caja.appendChild(cuerpo);
    return caja;
  }

  function tabla(serie) {
    var rollo = crea('div', 'rollo');
    rollo.appendChild(crea('h3', '', 'Valores comparados'));
    var t = crea('table', 'serie');
    var cabeza = crea('tr');
    var valores = crea('tr');
    serie.forEach(function (punto, i) {
      cabeza.appendChild(crea('th', '', fechaCorta(punto.semana)));
      valores.appendChild(crea('td', i === serie.length - 1 ? 'ultima' : '', punto.texto));
    });
    t.appendChild(cabeza);
    t.appendChild(valores);
    rollo.appendChild(t);
    return rollo;
  }

  // Small trend line of the compared weeks; gaps stay as gaps.
  function chispa(serie) {
    var ancho = 130, alto = 30, margen = 3;
    var lienzo = svg('svg', { 'class': 'chispa', width: ancho, height: alto, 'aria-hidden': 'true' });
    var numeros = serie.map(function (p) { return p.valor; }).filter(function (v) { return v !== null; });
    if (numeros.length < 2) return lienzo;
    var minimo = Math.min.apply(null, numeros), maximo = Math.max.apply(null, numeros);
    var rango = maximo - minimo || 1;
    var x = function (i) { return margen + i * (ancho - 2 * margen) / (serie.length - 1); };
    var y = function (v) { return alto - margen - (v - minimo) * (alto - 2 * margen) / rango; };
    var tramo = [];
    var cierra = function () {
      if (tramo.length > 1) {
        lienzo.appendChild(svg('polyline', { points: tramo.join(' '), fill: 'none', stroke: '#8f8782', 'stroke-width': 1.5 }));
      }
      tramo = [];
    };
    serie.forEach(function (punto, i) {
      if (punto.valor === null) { cierra(); return; }
      tramo.push(x(i).toFixed(1) + ',' + y(punto.valor).toFixed(1));
    });
    cierra();
    var ultimo = serie[serie.length - 1];
    if (ultimo.valor !== null) {
      lienzo.appendChild(svg('circle', { cx: x(serie.length - 1).toFixed(1), cy: y(ultimo.valor).toFixed(1), r: 3, fill: '#1f1d1d' }));
    }
    return lienzo;
  }

  id('actualizar').addEventListener('click', function () { leer(true); });
  id('lamina-anterior').addEventListener('click', function () { mueveLamina(-1); });
  id('lamina-siguiente').addEventListener('click', function () { mueveLamina(1); });
  arrastreLaminas();
  window.addEventListener('resize', pintaLamina);
  id('carrusel').addEventListener('keydown', function (e) {
    if (e.key === 'ArrowLeft') mueveLamina(-1);
    if (e.key === 'ArrowRight') mueveLamina(1);
  });
  id('filtro-mercado').addEventListener('change', function (e) { estado.mercado = e.target.value; pintaLista(); });
  id('filtro-sentido').addEventListener('change', function (e) { estado.sentido = e.target.value; pintaLista(); });
  id('filtro-texto').addEventListener('input', function (e) { estado.texto = e.target.value.trim().toLowerCase(); pintaLista(); });
  id('tab-movimientos').addEventListener('click', function () { estado.pestana = 'movimientos'; pintaLista(); });
  id('tab-calidad').addEventListener('click', function () { estado.pestana = 'calidad'; pintaLista(); });
  leer(false);
})();
